About This Demo

This page shows how to compose pages from reusable fragments.

Fragment file structure

templates/
  fragments.html              (main page)
  fragments/
    header.html               (th:fragment="header(title)")
    footer.html               (th:fragment="footer")
    content-overview.html     (th:fragment="content")
    content-customers.html    (th:fragment="content")
    content-about.html        (th:fragment="content")

Controller

@GetMapping("/fragments/{tab}")
String fragmentsWithTab(@PathVariable String tab, Model model) {
    model.addAttribute("activeTab", tab);
    model.addAttribute("customers", customerList());
    return "fragments";
}

// The main template uses th:switch on activeTab
// to decide which content fragment to include.
